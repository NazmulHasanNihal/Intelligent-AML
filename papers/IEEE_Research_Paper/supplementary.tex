\section{Algorithmic Procedures \& Pipeline Architecture}
\label{sec:algorithms}
This section formalizes the multi-stage training routine (Algorithm~\ref{alg:supp_cstgb_train}) and streaming online inference triage engine (Algorithm~\ref{alg:supp_cstgb_infer}) corresponding to the end-to-end execution flowchart detailed in Fig.~\ref{fig:supp_algorithm_flowchart}.

\begin{algorithm}[htbp]
\caption{C-STGB Unified Multi-Stage Training Pipeline}
\label{alg:supp_cstgb_train}
\begin{algorithmic}[1]
\Require Graph $\mathcal{G}=(\mathcal{V}, \mathcal{E})$, Features $\mathbf{X}$, Timestamps $t$, Target $\alpha$, Weights $\gamma_1=0.50, \gamma_2=0.10$.
\Ensure Trained model parameters $\Theta$, Calibrated threshold $\tau^*$, Conformal quantiles $\hat{q}^{(0)}, \hat{q}^{(1)}$.
\State Compute 12-D canonical invariants $\mathbf{z}_{\text{inv}}(u)$ for all $u \in \mathcal{V}$.
\State Cluster minority nodes into latent typologies $\mathcal{C}_1, \dots, \mathcal{C}_K$.
\State Synthesize minority representations $\mathbf{h}_{\text{syn}}$ and bilinear edges $\hat{\mathbf{A}}_{\text{syn}}$.
\State Mine hard-negative commercial hubs into batch bank $\mathcal{B}_{\text{hard}}$.
\For{epoch $= 1$ to $N_{\text{epochs}}$}
    \State Compute multi-scale temporal embeddings $\Phi_{\text{time}}(\Delta t_{ij})$ and weights $w_{ij}(t)$.
    \State Evaluate context-aware edge trust gates $\mathbf{g}_{ij}$ and filter camouflage links.
    \State Aggregate degree-capped temporal messages to obtain $\mathbf{h}_u^{(L)}$.
    \State Compute evidence-adaptive fusion weights $\alpha_u$ and representations $\mathbf{h}_u^*$.
    \State Evaluate loss $\mathcal{L}_{\text{total}} = \mathcal{L}_{\text{task}} + \gamma_1 \mathcal{L}_{\text{recon}} + \gamma_2 \mathcal{L}_{\text{aux}}$.
    \State Update $\Theta$ via AdamW with cosine learning rate annealing.
\EndFor
\State Optimize Bayes decision threshold $\tau^*$ on validation split $\mathcal{D}_{\text{val}}$.
\State Calibrate class-conditional non-conformity quantiles $\hat{q}^{(0)}, \hat{q}^{(1)}$ on $\mathcal{D}_{\text{cal}}$.
\State \textbf{Return} $\Theta, \tau^*, \hat{q}^{(0)}, \hat{q}^{(1)}$.
\end{algorithmic}
\end{algorithm}

\begin{algorithm}[htbp]
\caption{Streaming Online Inference \& Risk-Controlled Triage}
\label{alg:supp_cstgb_infer}
\begin{algorithmic}[1]
\Require Streaming transaction $(u, v, t, \mathbf{e}_{uv})$, Parameters $\Theta, \tau^*, \hat{q}^{(0)}, \hat{q}^{(1)}$.
\Ensure Operational Triage Action $\in \{\text{Auto-Block}, \text{Review Queue}, \text{Clear}\}$.
\State Ingest transaction event and update temporally restricted ego-graph ($t_e \le t$).
\State Compute canonical invariants $\mathbf{z}_{\text{inv}}(u)$ and edge trust gate $\mathbf{g}_{uv}$.
\State Extract temporal message embeddings $\mathbf{h}_u^{(L)}$.
\State Compute adaptive fusion weight $\alpha_u$ and fused representation $\mathbf{h}_u^*$.
\State Predict anomaly probability $\hat{p}_u = \sigma(\operatorname{MLP}(\mathbf{h}_u^*))$.
\State Construct class-conditional conformal prediction set $\Gamma(u) \subseteq \{0, 1\}$.
\If{$\Gamma(u) == \{1\}$}
    \State \textbf{Route to Tier 1:} Quarantined Asset Hold \& Compliance Decision Support Dossier.
\ElsIf{$\Gamma(u) == \{0, 1\}$}
    \State \textbf{Route to Tier 2:} Selective Abstention $\bot \to$ Compliance Officer Queue.
\Else
    \State \textbf{Route to Tier 3:} Straight-Through Clear.
\EndIf
\end{algorithmic}
\end{algorithm}

\section{Extended Mathematical Proofs \& Conformal Derivations}
\label{sec:extended_proofs}

\subsection{Dvoretzky-Kiefer-Wolfowitz (DKW) Finite-Sample Bound under TV Drift}
In continuous-time streaming financial networks, classical exchangeability is challenged by continuous-time temporal autocorrelation and non-stationary behavioral drift. As formulated in Section~III-I of the main manuscript~\cite{tibshirani2019conformal,barber2023conformal,gibbs2021adaptive}, C-STGB employs online Adaptive Conformal Inference (ACI) tracking and a rolling calibration split $\mathcal{D}_{\text{cal}}(t)$ over window $W_{\text{cal}}$ as the explicit mechanisms bounding total variation ($d_{\mathrm{TV}}$) error. Under locally bounded total variation drift $d_{\mathrm{TV}}(\mathcal{P}_t, \mathcal{P}_{t - W_{\text{cal}}}) \le \epsilon_{\text{drift}}$, the non-asymptotic class-conditional coverage error bound decomposes cleanly via maximal coupling into an environmental distribution shift component and a finite-sample empirical process concentration component:
\begin{equation}
\begin{aligned}
&\left| \mathbb{P}_{t}\left(Y \in \Gamma(X) \mid Y=y\right) - (1 - \alpha) \right| \\
&\quad \le d_{\mathrm{TV}}(\mathcal{P}_t, \mathcal{P}_{t - W_{\text{cal}}}) + \sup_{s} \left| F_{\text{cal}}^{(y)}(s) - \hat{F}_{\text{cal}}^{(y)}(s) \right|
\end{aligned}
\end{equation}
where $n(y) = |\mathcal{D}_{\text{cal}}^{(y)}(t)|$ denotes the class-conditional calibration sample size.

\begin{proof}
Let $F_t^{(y)}(s) = \mathbb{P}_t(S^{(y)} \le s)$ denote the true cumulative distribution function (CDF) of class-conditional non-conformity scores under the streaming test distribution $\mathcal{P}_t$, and let $F_{\text{cal}}^{(y)}(s) = \mathbb{P}_{\text{cal}}(S^{(y)} \le s)$ denote the true CDF under the rolling calibration distribution $\mathcal{P}_{t - W_{\text{cal}}}$. Let $\hat{F}_{\text{cal}}^{(y)}(s) = \frac{1}{n(y)} \sum_{i=1}^{n(y)} \mathbb{I}(s_i^{(y)} \le s)$ denote the empirical CDF computed over the rolling calibration split $\mathcal{D}_{\text{cal}}(t)$, where $n(y) = |\mathcal{D}_{\text{cal}}^{(y)}(t)|$.

By definition of the conformal quantile $\hat{q}^{(y)}$, the test coverage probability evaluates to $\mathbb{P}_t(Y \in \Gamma(X) \mid Y=y) = \mathbb{P}_t(S^{(y)} \le \hat{q}^{(y)}) = F_t^{(y)}(\hat{q}^{(y)})$. Applying the triangle inequality:
\begin{align}
\big| F_t^{(y)}(\hat{q}^{(y)}) - (1 - \alpha) \big| &\le \big| F_t^{(y)}(\hat{q}^{(y)}) - F_{\text{cal}}^{(y)}(\hat{q}^{(y)}) \big| \nonumber \\
&\quad + \big| F_{\text{cal}}^{(y)}(\hat{q}^{(y)}) - \hat{F}_{\text{cal}}^{(y)}(\hat{q}^{(y)}) \big| \nonumber \\
&\quad + \big| \hat{F}_{\text{cal}}^{(y)}(\hat{q}^{(y)}) - (1 - \alpha) \big|
\end{align}
By the maximal coupling characterization of total variation distance, for any Borel set $A \subseteq \mathbb{R}$:
\begin{equation}
\big| \mathbb{P}_t(S^{(y)} \in A) - \mathbb{P}_{\text{cal}}(S^{(y)} \in A) \big| \le d_{\mathrm{TV}}(\mathcal{P}_t, \mathcal{P}_{t - W_{\text{cal}}})
\end{equation}
Thus, setting $A = (-\infty, \hat{q}^{(y)}]$ yields $|F_t^{(y)}(\hat{q}^{(y)}) - F_{\text{cal}}^{(y)}(\hat{q}^{(y)})| \le d_{\mathrm{TV}}(\mathcal{P}_t, \mathcal{P}_{t - W_{\text{cal}}})$. 

Next, by the classical Dvoretzky-Kiefer-Wolfowitz (DKW) inequality (Massart, 1990) applied strictly to the i.i.d. draws comprising the calibration split $\mathcal{D}_{\text{cal}}^{(y)}$:
\begin{equation}
\mathbb{P}\left( \sup_{s \in \mathbb{R}} \big| F_{\text{cal}}^{(y)}(s) - \hat{F}_{\text{cal}}^{(y)}(s) \big| \le \sqrt{\frac{\ln(2/\delta)}{2 n(y)}} \;\middle\vert\; Y=y \right) \ge 1 - \delta
\end{equation}
Finally, by the definition of the discrete empirical quantile with finite sample size $n(y)$, the discretization residual satisfies $|\hat{F}_{\text{cal}}^{(y)}(\hat{q}^{(y)}) - (1 - \alpha)| \le \frac{1}{n(y)}$. Under non-atomic score continuity ($L_s < \infty$), combining these terms establishes the non-asymptotic finite-sample bound with high probability $1 - \delta$. Crucially, this confirms that distribution shift $d_{\mathrm{TV}}$ is an external environmental property tracked via rolling calibration and ACI, whereas sampling uncertainty decays at $\mathcal{O}(1/\sqrt{n(y)})$.
\end{proof}

\subsection{Chronological Temporal Purity and Zero-Leakage of Latent GraphSMOTE}
\begin{lemma}[Chronological Isolation of Latent Topological Oversampling]
\label{lem:graphsmote_causality}
Let $\mathcal{D}_{\text{train}} = \mathcal{G}_{[0, t_{\text{train}}]}$, $\mathcal{D}_{\text{val}} = \mathcal{G}_{(t_{\text{train}}, t_{\text{val}}]}$, $\mathcal{D}_{\text{cal}} = \mathcal{G}_{(t_{\text{val}}, t_{\text{cal}}]}$, and $\mathcal{D}_{\text{test}} = \mathcal{G}_{(t_{\text{cal}}, t_{\text{test}}]}$ denote a strictly chronological 4-way temporal partition. Let $\mathbf{h}_{\text{syn}}$ be any synthetic minority representation generated via convex combination of seed nodes $u, v \in \mathcal{C}_k \subseteq \mathcal{V}_{\text{train}}$ with birth timestamp $t_{\text{syn}} = \max(t_u, t_v) \le t_{\text{train}}$. Under the causal indicator constraint $\hat{\mathbf{A}}_{\text{syn}, w} = \sigma(\mathbf{h}_{\text{syn}}^T \mathbf{W}_{\text{edge}} \mathbf{h}_w) \cdot \mathbb{I}(t_w \le t_{\text{syn}})$, the synthesized graph $\tilde{\mathcal{G}_{\text{train}}}$ satisfies:
\begin{equation}
\begin{aligned}
&\mathcal{V}(\tilde{\mathcal{G}}_{\text{train}}) \cap \big(\mathcal{V}_{\text{val}} \cup \mathcal{V}_{\text{cal}} \cup \mathcal{V}_{\text{test}}\big) = \emptyset, \quad \text{and} \\
&\forall (i, j) \in \mathcal{E}(\tilde{\mathcal{G}}_{\text{train}}), \quad \max(t_i, t_j) \le t_{\text{train}}
\end{aligned}
\end{equation}
Consequently, synthetic edge formation induces zero forward lookahead bias and zero backpropagation gradient flow into prospective evaluation partitions.
\end{lemma}
\begin{proof}
By construction, parent seed nodes $u, v \in \mathcal{V}_{\text{train}}$, which implies $t_u, t_v \le t_{\text{train}}$ and hence $t_{\text{syn}} \le t_{\text{train}}$. Candidate target nodes $w$ are strictly restricted to $\mathcal{C}_{\text{cand}}(\mathbf{h}_{\text{syn}}) \cap \mathcal{V}_{\text{train}}^{\le t_{\text{syn}}}$. The causal indicator $\mathbb{I}(t_w \le t_{\text{syn}})$ evaluates to 0 for any node with $t_w > t_{\text{syn}}$, enforcing $\hat{\mathbf{A}}_{\text{syn}, w} = 0$. Because $t_{\text{syn}} \le t_{\text{train}} < t_{\text{val}} < t_{\text{cal}} < t_{\text{test}}$, no candidate $w \in \mathcal{V}_{\text{val}} \cup \mathcal{V}_{\text{cal}} \cup \mathcal{V}_{\text{test}}$ can satisfy $t_w \le t_{\text{syn}}$. Furthermore, synthetic node representations $\mathbf{h}_{\text{syn}}$ and virtual edges are discarded prior to evaluation on $\mathcal{D}_{\text{val}}, \mathcal{D}_{\text{cal}}$, and $\mathcal{D}_{\text{test}}$. Hence, mutual information satisfies $I(\tilde{\mathcal{G}}_{\text{train}}; \mathcal{D}_{\text{test}} \mid \mathcal{D}_{\text{train}}) = 0$.
\end{proof}

\subsection{Kirchhoff Mass-Balance Flow Conservation Invariant for Layering Cycles}
\begin{lemma}[Conservation of Flow in Pass-Through Conduit Chains]
\label{lem:flow_conservation}
Let $\mathcal{P}_{\text{chain}} = (v_0, v_1, \dots, v_K)$ denote an ordered multi-hop fund layering chain with transaction amounts $A(v_{k-1}, v_k)$ and retention loss rate $\epsilon_k \in [0, 1)$ at each hop (representing service fees or split commissions). If intermediate nodes $v_k$ ($1 \le k < K$) act strictly as pass-through conduit accounts without capital injection, then the aggregate flow conservation invariant satisfies:
\begin{equation}
\Phi_{\text{flow}}(v_k, t) = \frac{\sum_{e \in \mathcal{E}_{\text{out}}(v_k)} A(e)}{\sum_{e \in \mathcal{E}_{\text{in}}(v_k)} A(e) + \epsilon} \in \left[ 1 - \bar{\epsilon}, 1.0 \right]
\end{equation}
where $\bar{\epsilon} = \max_k \epsilon_k < 0.15$. Consequently, the 12-D canonical feature $\tilde{\Phi}_{\text{flow}}(v_k, t) = \log(1 + \Phi_{\text{flow}}(v_k, t))$ is strictly bounded within $[\log(2 - \bar{\epsilon}), \log(2)] \approx [0.615, 0.693]$, forming an invariant topological signature that is invariant to arbitrary graph scale and entity degree.
\end{lemma}
\begin{proof}
By the definition of intermediate pass-through mules in financial layering (BSA 31 CFR \S 1010.100), the net retained capital $R(v_k) = \sum_{\text{in}} A - \sum_{\text{out}} A$ is bounded exclusively by transaction clearing fees $\epsilon_k \sum_{\text{in}} A$. Thus $\sum_{\text{out}} A = (1 - \epsilon_k) \sum_{\text{in}} A$. Dividing by $\sum_{\text{in}} A$ yields $\Phi_{\text{flow}}(v_k, t) = 1 - \epsilon_k$. Since $\epsilon_k \in [0, \bar{\epsilon}]$, the ratio $\Phi_{\text{flow}}(v_k, t)$ lies in $[1 - \bar{\epsilon}, 1.0]$. Applying the monotonic logarithmic transformation $\tilde{\Phi}_{\text{flow}} = \log(1 + \Phi_{\text{flow}})$ preserves the interval order, yielding the bounded invariant range.
\end{proof}

\begin{proposition}[Convex Combination Flow Invariant Preservation (Proposition S2)]
\label{prop:graphsmote_flow_preservation}
Let $u, v \in \mathcal{C}_k$ be two parent pass-through accounts in the same latent typology cluster with total inflows $I_u, I_v > 0$ and outflows $O_u = \Phi_u I_u, O_v = \Phi_v I_v$, where $\Phi_u, \Phi_v \in [1 - \bar{\epsilon}, 1.0]$. Let the synthetic representation $\mathbf{h}_{\text{syn}} = \lambda \mathbf{h}_u + (1 - \lambda) \mathbf{h}_v$ ($\lambda \in [0, 1]$) linearly interpolate parent flow amounts, such that $I_{\text{syn}} = \lambda I_u + (1 - \lambda) I_v$ and $O_{\text{syn}} = \lambda O_u + (1 - \lambda) O_v$. Then the synthetic flow conservation ratio $\Phi_{\text{flow}}(\mathbf{h}_{\text{syn}}) = \frac{O_{\text{syn}}}{I_{\text{syn}}}$ satisfies:
\begin{equation}
\min(\Phi_u, \Phi_v) \le \Phi_{\text{flow}}(\mathbf{h}_{\text{syn}}) \le \max(\Phi_u, \Phi_v)
\end{equation}
In particular, $\Phi_{\text{flow}}(\mathbf{h}_{\text{syn}}) \in [1 - \bar{\epsilon}, 1.0]$.
\end{proposition}
\begin{proof}
Without loss of generality, assume $\Phi_u \le \Phi_v$. Since $\lambda \in [0, 1]$ and $I_u, I_v > 0$, we have:
\begin{align}
O_{\text{syn}} &= \lambda \Phi_u I_u + (1 - \lambda) \Phi_v I_v \nonumber \\
&\ge \lambda \Phi_u I_u + (1 - \lambda) \Phi_u I_v = \Phi_u I_{\text{syn}}, \nonumber \\
O_{\text{syn}} &= \lambda \Phi_u I_u + (1 - \lambda) \Phi_v I_v \nonumber \\
&\le \lambda \Phi_v I_u + (1 - \lambda) \Phi_v I_v = \Phi_v I_{\text{syn}}.
\end{align}
Dividing throughout by $I_{\text{syn}} > 0$ yields $\Phi_u \le \frac{O_{\text{syn}}}{I_{\text{syn}}} \le \Phi_v$. Because $\Phi_u, \Phi_v \in [1 - \bar{\epsilon}, 1.0]$, it follows directly that $\Phi_{\text{flow}}(\mathbf{h}_{\text{syn}}) \in [1 - \bar{\epsilon}, 1.0]$. Hence, convex interpolation strictly preserves the bounded flow conservation property within the parent convex hull without violating monetary mass-balance bounds.
\end{proof}

\section{14-Dataset Invariant-Augmented GNN Cross-Evaluation Matrix}
\label{sec:supp_invariant_gnn}

Table~\ref{tab:supp_invariant_gnn_crosseval} reports the complete multi-dataset performance breakdown across all 14 benchmark networks for spatial and dynamic GNN architectures (GCN, GraphSAGE, and EvolveGCN) trained with canonical node attributes concatenated with the full 12-dimensional domain invariant vector $[\mathbf{x}_u \,\|\, \mathbf{z}_{\text{inv}}(u, t)]$.

\section{Topological Explanation Fidelity, Factual Sparsity, and Counterfactual Validation}
\label{sec:supp_xai_fidelity}

To substantiate model explainability under regulatory scrutiny (such as Federal Reserve SR 26-2~\cite{fed2026sr262} and EU AI Act Article 13 transparency mandates), this section formalizes the quantitative attribution fidelity, explanation sparsity, and counterfactual validation of C-STGB's GNNExplainer topological extraction pipeline, evaluated across $N=500$ confirmed illicit transaction clusters from \texttt{elliptic\_v1}, \texttt{paysim\_extended}, and \texttt{saml\_d} (Table~\ref{tab:supp_xai_fidelity}).

\subsection{Mathematical Definitions of Attribution Metrics}
\begin{enumerate}
    \item \textbf{Factual Subgraph Fidelity ($\text{Fid}^+$):} Quantifies whether the isolated causal subgraph $\mathcal{G}_s$ captures the sufficient and necessary relational topology triggering the anomaly classification:
    \begin{equation}
    \text{Fid}^+ = \frac{1}{N} \sum_{i=1}^N \left( \hat{p}(y_i=1 \mid \mathcal{G}_i) - \hat{p}(y_i=1 \mid \mathcal{G}_i \setminus \mathcal{G}_{s,i}) \right)
    \end{equation}
    A high $\text{Fid}^+$ score ($0.942$) demonstrates that removing the salient subgraph collapses illicit confidence to baseline benign levels.
    
    \item \textbf{Counterfactual Infidelity ($\text{Fid}^-$):} Measures whether background chaff edges outside the explanation mask contain residual illicit information:
    \begin{equation}
    \text{Fid}^- = \frac{1}{N} \sum_{i=1}^N \left( \hat{p}(y_i=1 \mid \mathcal{G}_i) - \hat{p}(y_i=1 \mid \mathcal{G}_{s,i}) \right)
    \end{equation}
    A near-zero $\text{Fid}^-$ score ($0.018$) confirms that the isolated subgraph alone preserves full model prediction accuracy without uncaptured structural residuals.
    
    \item \textbf{Explanation Sparsity:} Measures the fraction of the $2$-hop computation graph pruned as commercial or benign noise:
    \begin{equation}
    \text{Sparsity} = 1 - \frac{|\mathcal{E}_s|}{|\mathcal{E}_c|}
    \end{equation}
    where $|\mathcal{E}_c|$ is the computational ego-graph size and $|\mathcal{E}_s|$ is the pruned causal conduit ($86.5\%$ average compression).
    
    \item \textbf{Counterfactual Validity:} The empirical rate at which normalizing attributed transaction amounts or diluting burst frequency successfully transitions borderline or high-risk accounts from Tier~3 to Tier~1 ($\Gamma(X) = \{0\}$), establishing actionable operational recourse.
\end{enumerate}
