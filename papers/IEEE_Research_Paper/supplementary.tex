\section{Algorithmic Procedures \& Pipeline Architecture}
\label{sec:algorithms}
This section formalizes the multi-stage training routine (Algorithm~S1) and streaming online inference triage engine (Algorithm~S2) corresponding to the end-to-end execution flowchart detailed in Fig.~S1.

\begin{algorithm}[htbp]
\caption{C-STGB Unified Multi-Stage Training Pipeline (Algorithm S1)}
\label{alg:supp_cstgb_train}
\begin{algorithmic}[1]
\Require Graph $\mathcal{G}=(\mathcal{V}, \mathcal{E})$, Features $\mathbf{X}$, Timestamps $t$, Target $\alpha$, Weights $\gamma_1=0.50, \gamma_2=0.10$.
\Ensure Trained model parameters $\Theta$, Calibrated threshold $\tau^*$, Conformal quantiles $\hat{q}^{(0)}, \hat{q}^{(1)}$.
\State Compute 12-D canonical invariants $\mathbf{z}_{\text{inv}}(u)$ for all $u \in \mathcal{V}$.
\State Cluster minority nodes into latent typologies $\mathcal{C}_1, \dots, \mathcal{C}_K$.
\State Synthesize minority representations $\mathbf{h}_{\text{syn}}$ and bilinear edges $\hat{\mathbf{A}}_{\text{syn}}$.
\State Mine hard-negative commercial hubs into batch bank $\mathcal{B}_{\text{hard}}$.
\For{epoch $= 1$ to $N_{\text{epochs}}$}
    \State Compute multi-scale temporal embeddings $\Phi_{\text{time}}(\Delta t_{ij})$ and weights $w_{ij}(t)$.
    \State Evaluate context-aware edge trust gates $\mathbf{g}_{ij}$ and filter camouflage links.
    \State Aggregate degree-capped temporal messages to obtain $\mathbf{h}_u^{(L)}$.
    \State Compute evidence-adaptive fusion weights $\alpha_u$ and representations $\mathbf{h}_u^*$.
    \State Evaluate loss $\mathcal{L}_{\text{total}} = \mathcal{L}_{\text{task}} + \gamma_1 \mathcal{L}_{\text{recon}} + \gamma_2 \mathcal{L}_{\text{aux}}$.
    \State Update $\Theta$ via AdamW with cosine learning rate annealing.
\EndFor
\State Optimize Bayes decision threshold $\tau^*$ on validation split $\mathcal{D}_{\text{val}}$.
\State Calibrate class-conditional non-conformity quantiles $\hat{q}^{(0)}, \hat{q}^{(1)}$ on $\mathcal{D}_{\text{cal}}$.
\State \textbf{Return} $\Theta, \tau^*, \hat{q}^{(0)}, \hat{q}^{(1)}$.
\end{algorithmic}
\end{algorithm}

\begin{algorithm}[htbp]
\caption{Streaming Online Inference \& Risk-Controlled Triage (Algorithm S2)}
\label{alg:supp_cstgb_infer}
\begin{algorithmic}[1]
\Require Streaming transaction $(u, v, t, \mathbf{e}_{uv})$, Parameters $\Theta, \tau^*, \hat{q}^{(0)}, \hat{q}^{(1)}$.
\Ensure Operational Triage Action $\in \{\text{Auto-Block}, \text{Review Queue}, \text{Clear}\}$.
\State Ingest transaction event and update causal ego-graph ($t_e \le t$).
\State Compute canonical invariants $\mathbf{z}_{\text{inv}}(u)$ and edge trust gate $\mathbf{g}_{uv}$.
\State Extract temporal message embeddings $\mathbf{h}_u^{(L)}$.
\State Compute adaptive fusion weight $\alpha_u$ and fused representation $\mathbf{h}_u^*$.
\State Predict anomaly probability $\hat{p}_u = \sigma(\operatorname{MLP}(\mathbf{h}_u^*))$.
\State Construct class-conditional conformal prediction set $\Gamma(u) \subseteq \{0, 1\}$.
\If{$\Gamma(u) == \{1\}$}
    \State \textbf{Route to Tier 1:} Automated Quarantined Asset Hold \& SAR Draft.
\ElsIf{$\Gamma(u) == \{0, 1\}$}
    \State \textbf{Route to Tier 2:} Selective Abstention $\bot \to$ Compliance Officer Queue.
\Else
    \State \textbf{Route to Tier 3:} Straight-Through Clear.
\EndIf
\end{algorithmic}
\end{algorithm}

\section{Extended Mathematical Proofs \& Conformal Derivations}
\label{sec:extended_proofs}

\subsection{Dvoretzky-Kiefer-Wolfowitz (DKW) Finite-Sample Bound under TV Drift}
As formulated in Section~III-H of the main manuscript~\cite{tibshirani2019conformal,barber2023conformal,gibbs2021adaptive}, the non-asymptotic class-conditional coverage error bound under locally bounded total variation drift $TV(\mathcal{P}_t, \mathcal{P}_{t - W_{\text{cal}}}) \le \epsilon_{\text{drift}}$ is given by:
\begin{equation}
\left| \mathbb{P}_{t}\left(Y \in \Gamma(X) \mid Y=y\right) - (1 - \alpha) \right| \le 2 \cdot TV(\mathcal{P}_t, \mathcal{P}_{t - W_{\text{cal}}}) + \frac{C_{\text{score}}}{\sqrt{|\mathcal{D}_{\text{cal}}(t)|}}
\end{equation}
where $C_{\text{score}} = \sqrt{\frac{1}{2}\ln(2/\delta)}$ is the Dvoretzky-Kiefer-Wolfowitz (DKW) empirical process constant at confidence level $1-\delta$.

\begin{proof}
Let $F_t^{(y)}(s) = \mathbb{P}_t(S^{(y)} \le s)$ denote the true cumulative distribution function (CDF) of non-conformity scores under the test distribution $\mathcal{P}_t$, and let $\hat{F}_{\text{cal}}^{(y)}(s) = \frac{1}{n(y)} \sum_{i=1}^{n(y)} \mathbb{I}(s_i^{(y)} \le s)$ denote the empirical CDF computed over the rolling calibration split $\mathcal{D}_{\text{cal}}(t)$, where $n(y) = \rho_y |\mathcal{D}_{\text{cal}}(t)|$ and $\rho_y > 0$ denotes the calibration prevalence of class $y$. By the triangle inequality:
\begin{align}
\sup_s \left| F_t^{(y)}(s) - \hat{F}_{\text{cal}}^{(y)}(s) \right| &\le \sup_s \left| F_t^{(y)}(s) - F_{\text{cal}}^{(y)}(s) \right| \nonumber \\
&\quad + \sup_s \left| F_{\text{cal}}^{(y)}(s) - \hat{F}_{\text{cal}}^{(y)}(s) \right|
\end{align}
By coupling and total variation definition, $\sup_s |F_t^{(y)}(s) - F_{\text{cal}}^{(y)}(s)| \le 2 \cdot TV(\mathcal{P}_t, \mathcal{P}_{t - W_{\text{cal}}})$. By the classical DKW inequality for empirical processes, for any $\delta \in (0, 1)$:
\begin{equation}
\mathbb{P}\left( \sup_s \left| F_{\text{cal}}^{(y)}(s) - \hat{F}_{\text{cal}}^{(y)}(s) \right| \le \sqrt{\frac{\ln(2/\delta)}{2 n(y)}} \right) \ge 1 - \delta
\end{equation}
Setting $C_{\text{score}} = \sqrt{\frac{1}{2}\ln(2/\delta)}$ and observing that $n(y) = \rho_y |\mathcal{D}_{\text{cal}}(t)|$ establishes the finite-sample bound under non-atomic score regularity ($L_s < \infty$).
\end{proof}
